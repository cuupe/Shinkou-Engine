cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(ShinkouCore VERSION 0.1.0 LANGUAGES CXX)
    include(CTest)
endif()
if(TARGET shinkou_core)
    return()
endif()
add_library(shinkou_core STATIC src/Error.cpp src/Json.cpp)
add_library(ShinkouEngine::Core ALIAS shinkou_core)
target_compile_features(shinkou_core PUBLIC cxx_std_17)
target_include_directories(shinkou_core PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/ShinkouFetchDependency.cmake")
shinkou_fetch_dependency(json json_SOURCE_DIR)
target_include_directories(shinkou_core SYSTEM PRIVATE "${json_SOURCE_DIR}/single_include")
if(MSVC)
    target_compile_options(shinkou_core PRIVATE /W4 /permissive-)
else()
    target_compile_options(shinkou_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING AND (NOT DEFINED SHINKOU_BUILD_TESTS OR SHINKOU_BUILD_TESTS))
    add_executable(shinkou_core_tests tests/CoreTests.cpp)
    target_link_libraries(shinkou_core_tests PRIVATE shinkou_core)
    add_test(NAME shinkou_core_tests COMMAND shinkou_core_tests)
endif()
install(TARGETS shinkou_core ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
shinkou_install_dependency_licenses(json "${json_SOURCE_DIR}")
